cmake_minimum_required(VERSION 3.16)
project(r2b4_nav2_wheel_adapter LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_util REQUIRED)
find_package(pluginlib REQUIRED)
add_library(r2b4_nav2_wheel_adapter SHARED controller.cpp)
target_link_directories(r2b4_nav2_wheel_adapter PRIVATE "${R2B4_NAV2_PREFIX}/lib")
ament_target_dependencies(r2b4_nav2_wheel_adapter nav2_regulated_pure_pursuit_controller
  nav2_core nav2_costmap_2d nav2_util pluginlib)
target_link_libraries(r2b4_nav2_wheel_adapter
  "${R2B4_NAV2_PREFIX}/lib/libnav2_regulated_pure_pursuit_controller.so")
set_target_properties(r2b4_nav2_wheel_adapter PROPERTIES BUILD_RPATH "${R2B4_NAV2_PREFIX}/lib")
install(TARGETS r2b4_nav2_wheel_adapter LIBRARY DESTINATION lib)
pluginlib_export_plugin_description_file(nav2_core plugin.xml)
ament_package()
